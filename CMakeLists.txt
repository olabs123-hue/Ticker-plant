cmake_minimum_required(VERSION 3.16)
project(tickerplant LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tickerplant src/tickerplant.cpp)
target_include_directories(tickerplant PUBLIC include)
target_compile_options(tickerplant PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tickerplant_bench benchmarks/benchmark.cpp)
target_link_libraries(tickerplant_bench PRIVATE tickerplant)
include(CTest)
if(BUILD_TESTING)
 include(FetchContent)
 FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
 FetchContent_MakeAvailable(googletest)
 add_executable(tickerplant_tests tests/test_tickerplant.cpp)
 target_link_libraries(tickerplant_tests PRIVATE tickerplant GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(tickerplant_tests)
endif()
